## Database Tables Required

```sql
-- Type/Tense Master
CREATE TABLE grammar_type_details (
  id SERIAL PRIMARY KEY,
  type_id INT REFERENCES grammar_types(id),
  deep_explanation TEXT,
  formation_structure TEXT,
  metadata JSONB,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Examples Table
CREATE TABLE type_examples (
  id SERIAL PRIMARY KEY,
  type_id INT REFERENCES grammar_types(id),
  sentence TEXT,
  translation TEXT,
  noun_breakdown JSONB,
  context TEXT,
  difficulty VARCHAR(20),
  created_at TIMESTAMP DEFAULT NOW()
);

-- Grammar Rules Table
CREATE TABLE type_grammar_rules (
  id SERIAL PRIMARY KEY,
  type_id INT REFERENCES grammar_types(id),
  rule_name VARCHAR(255),
  explanation TEXT,
  do_example TEXT,
  dont_example TEXT,
  key_points JSONB,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Common Mistakes Table
CREATE TABLE type_common_mistakes (
  id SERIAL PRIMARY KEY,
  type_id INT REFERENCES grammar_types(id),
  mistake TEXT,
  correction TEXT,
  explanation TEXT,
  rule_violated VARCHAR(255),
  created_at TIMESTAMP DEFAULT NOW()
);

-- Writing Exercise Table
CREATE TABLE type_writing_exercises (
  id SERIAL PRIMARY KEY,
  type_id INT REFERENCES grammar_types(id),
  exercise_type VARCHAR(50),
  prompt TEXT,
  guidelines JSONB,
  sample_answer TEXT,
  evaluation_criteria JSONB,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Reading Comprehension Table
CREATE TABLE type_reading_comprehension (
  id SERIAL PRIMARY KEY,
  type_id INT REFERENCES grammar_types(id),
  passage_title VARCHAR(255),
  passage_text TEXT,
  highlighted_items JSONB,
  questions JSONB,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Video Resources Table
CREATE TABLE type_video_resources (
  id SERIAL PRIMARY KEY,
  type_id INT REFERENCES grammar_types(id),
  video_title VARCHAR(255),
  video_url VARCHAR(500),
  description TEXT,
  duration VARCHAR(20),
  difficulty VARCHAR(20),
  key_topics JSONB,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Practice Questions Table
CREATE TABLE type_practice_questions (
  id SERIAL PRIMARY KEY,
  type_id INT REFERENCES grammar_types(id),
  difficulty_level VARCHAR(20),
  question_type VARCHAR(50),
  question TEXT,
  options JSONB,
  correct_answer VARCHAR(255),
  explanation TEXT,
  created_at TIMESTAMP DEFAULT NOW()
);
```

